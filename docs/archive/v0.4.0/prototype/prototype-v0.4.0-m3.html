系统架构与偏好设置热重载验证

mditor 致力于提供纯粹的打字体验。在本版本中，偏好设置中的字号、行宽限制与行距将即时作用于整个编辑画布，无需重启应用。

「简单至极，即是终极的精纯与力量。」 
export function applyPreferences(prefs: EditorPreferences) {
  document.documentElement.style.setProperty('--editor-font-size', `${prefs.fontSize}px`);
  document.documentElement.style.setProperty('--editor-max-width', `${prefs.maxWidth}px`);
}
